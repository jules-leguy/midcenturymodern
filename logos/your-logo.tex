% Logo shipped with the demo: the theme's own mark, "midcentury / modern", for
% the user to replace with theirs. The file names keep "your-logo" so the
% \titlegraphic line in demo.tex says as much.
% Two pages: 1 = Kraft (terracotta), 2 = DeepBlue (navy).
%
% Drawn from primitives, nothing traced: two orbits crossed at +/-45 degrees
% around a small nucleus, over a pale disc printed off register, as in a
% two-colour job. Each orbit is the ring between an ellipse and a smaller one
% shifted inside it, so the stroke swells along one side and thins along the
% other. The wordmark is lowercase TeX Gyre Adventor, the theme's title face,
% on two lines.
%
% Chosen by Jules on 2026-10-20 out of the study in local_testing/logo-atom/
% (mark 9 with align.pdf p.2, raised 0.25 mm: raise.pdf p.2).
%
% Deliberate choices, all settled by rendering at final size:
%   - Orbits at +/-45, not +/-30: at 30 degrees the two orbits merge into an
%     almond and the nucleus into a pupil, and the mark reads as an eye.
%   - The text is centred on the nucleus by its INK, not by its box, then
%     raised 0.25 mm (\mcmwmrise). The box keeps room under "modern" for a
%     descender it does not have, so box-centred text sits 0.7 mm high.
%     Ink extents come from the boxes of the two lines (glyph-true heights
%     under luaotfload), so changing the words keeps them centred.
%   - The thin side of an orbit is 0.23 in the drawing's own units (0.55 on
%     the thick side), i.e. 0.21 mm once placed: just above the 0.20 mm floor
%     under which a line falls under the footer strip's own scan lines.
%
% Rebuild:
%   lualatex your-logo.tex
%   pdfseparate your-logo.pdf tmp-%d.pdf
%   mv tmp-1.pdf your-logo-kraft.pdf ; mv tmp-2.pdf your-logo-deepblue.pdf
%   rm -f your-logo.aux your-logo.log your-logo.pdf
%
% Sizes are final: the artwork is 8.8 mm tall, meant to be placed with
% \includegraphics[height=0.8cm]. The theme measures whatever \titlegraphic is
% handed, so the gap in the scan lines follows the logo's width on its own.
\documentclass[multi=tikzpicture,border=0.4pt]{standalone}
\usepackage{fontspec}
\usepackage{tikz}
\definecolor{mcmBg}{HTML}{F8F6F2}
\definecolor{kraftink}{HTML}{9C4732}
\definecolor{blueink}{HTML}{002D4D}
\newfontfamily\mcmwordmark{TeX Gyre Adventor}[LetterSpace=-1.0]
\newcommand{\mcmwmfont}{\mcmwordmark\fontsize{12.5}{12.5}\selectfont}
\newcommand{\mcmwmone}{midcentury}
\newcommand{\mcmwmtwo}{modern}
\newcommand{\mcmwmrise}{0.25}% mm the text sits above true centre
\newcommand{\mcmwmx}{5.2}%     mm from the nucleus to the text

% One orbit, tilted by #2: the ring between two ellipses.
\newcommand{\mcmorbit}[2]{%
  \fill[#1,even odd rule,rotate=#2]
    (0,0) ellipse (4.35 and 1.95) (0,0.16) ellipse (4.08 and 1.56);}

% Measuring the two lines. This MUST run outside any tikzpicture: an \sbox
% inside one, under standalone[multi=tikzpicture], comes back zero-width.
% \mcmL is the baseline-to-baseline gap; \mcmB the first line's baseline, so
% that the ink (top of line 1 to bottom of line 2) is centred on 0, then
% raised.
\newsavebox{\mcmbxa}\newsavebox{\mcmbxb}
\newcommand{\mcmfit}{%
  \sbox\mcmbxa{\mcmwmfont\mcmwmone}\sbox\mcmbxb{\mcmwmfont\mcmwmtwo}%
  \pgfmathsetmacro{\mcmL}{3.106}%
  \pgfmathsetmacro{\mcmB}{(\mcmL+((\the\dp\mcmbxb)-(\the\ht\mcmbxa))/2.84528)/2
                          +\mcmwmrise}}

\newcommand{\mcmlogo}[1]{%
  \begin{tikzpicture}[x=1mm,y=1mm]
    \path (0,-4.4) (0,4.4);                    % common box, 8.8 mm tall
    \fill[#1!20!mcmBg] (0.7,-0.5) circle (3.5);
    \mcmorbit{#1}{45}\mcmorbit{#1}{135}
    \fill[#1] (0,0) circle (0.6);
    \node[anchor=base west,inner sep=0,text=#1,font=\mcmwmfont]
      at (\mcmwmx,\mcmB) {\mcmwmone};
    \node[anchor=base west,inner sep=0,text=#1,font=\mcmwmfont]
      at (\mcmwmx,\mcmB-\mcmL) {\mcmwmtwo};
  \end{tikzpicture}}

\begin{document}
\mcmfit
\mcmlogo{kraftink}
\mcmlogo{blueink}
\end{document}
