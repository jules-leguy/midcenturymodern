% =============================================================================
% titlepage.tex — QA deck for the title page template.
%
% One frame per title-page case, exercising very different amounts of text in
% every metadata field (title, subtitle, author, institute, date, logo) so
% that regressions in \setbeamertemplate{title page} show up immediately.
%
% Every name, institution and event below is an invented placeholder, chosen
% only for its length and shape. The proper nouns are coinages that were each
% checked (web search, 2026-08-19) not to match an existing organisation, place
% or notable person — Thalmire, Quernhaven, Kirnwald, Sorenfeld, Draventhal,
% Belmoncourt, Calvenne, Quenvard, and the surnames Rennholt, Vaudrenne,
% Odumbaya, Kellwig, Ferrandin, Fennrick, Delmarnier, Marquenot, Anjaveli,
% Estvik, Ravikoti, Ivelen, Brissenot, Hendral, Ångfeldt. Keep it that way when
% adding cases: a plausible-sounding real-world name ("Meridian Analytics",
% "Lakeside Polytechnic") almost always turns out to belong to someone.
%
% Compile from this directory, with the theme on the input path:
%   TEXINPUTS=..: lualatex titlepage.tex
% Twice: `remember picture' needs a second pass (on the first one every
% `current page' node collapses to the origin). No biber needed.
% =============================================================================
\documentclass[aspectratio=169]{beamer}
\usepackage{beamerthememidcenturymodern}
\graphicspath{{../logos/}}

% --- Case label: a tiny marker in the top-left corner, well away from the
%     right-aligned title block, so each page is identifiable in review.
\newcommand{\mcmCase}[1]{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north west, inner sep=0pt, text=mcmPrimaryFg!60]
      at ([xshift=0.25cm, yshift=-0.2cm]current page.north west)
      {\tiny\ttfamily #1};
  \end{tikzpicture}%
}

\newcommand{\casepage}[1]{%
  \begin{frame}
    \titlepage
    \mcmCase{#1}
  \end{frame}}
