\begin{frame}{Two Equal Columns}

        \begin{itemize}
          \item Manual review of every submission, one at a time
          \item Feedback collected in a shared spreadsheet
          \item Turnaround time varies widely between reviewers
          \item No automated check before the manual pass
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Proposed Approach}
        \begin{itemize}
          \item Automated pre-checks catch obvious issues first
          \item Reviewers focus only on borderline cases

          \item Turnaround time bounded by a fixed review window
        \end{itemize}

\begin{frame}{2$\times$2 Block Grid}

        else is already doing.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}

      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{columns}[T]

        maintenance savings elsewhere in the system.}

    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{High Cost, High Risk}
        {\small Reserved for the rare case where the upside clearly
        outweighs both the spend and the chance of failure.}

\begin{frame}{Definition List}
  \begin{description}
    \item[Throughput] The number of requests the system processes per
      unit of time, regardless of how long any single one takes.
    \item[Latency] The time a single request takes from submission to
      response, independent of how many others run alongside it.
    \item[Availability] The fraction of time the system is able to
      serve requests at all, usually tracked over a rolling window.
    \item[Error rate] The fraction of requests that fail, expressed as
      a percentage of total traffic.
  \end{description}
\end{frame}
