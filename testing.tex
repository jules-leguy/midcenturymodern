\section{Progressive Reveals}
% =====================================================================

\begin{frame}{Build with \texttt{\textbackslash pause}}
  \begin{itemize}
    \item First point appears immediately \pause
    \item Second point appears on the next click \pause
    \item Third point appears last
  \end{itemize}
\end{frame}

\begin{frame}{Revealing a Column with \texttt{\textbackslash onslide}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Always Visible}
        {\small This column is shown from the first click.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \onslide<2->{%
        \begin{alertblock}{Revealed on Click 2}
          {\small This column only appears on the second overlay step.}
        \end{alertblock}
      }
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Progressive Emphasis Across Blocks}
  \begin{columns}[T]
    \begin{column}{0.31\textwidth}
      \only<1>{\begin{alertblock}{Step 1}{\small Currently active}\end{alertblock}}%
      \only<2->{\begin{block}{Step 1}{\small Completed}\end{block}}%
    \end{column}
    \begin{column}{0.31\textwidth}
      \only<1>{\begin{block}{Step 2}{\small Not yet started}\end{block}}%
      \only<2>{\begin{alertblock}{Step 2}{\small Currently active}\end{alertblock}}%
      \only<3->{\begin{block}{Step 2}{\small Completed}\end{block}}%
    \end{column}
    \begin{column}{0.31\textwidth}
      \only<1-2>{\begin{block}{Step 3}{\small Not yet started}\end{block}}%
      \only<3>{\begin{alertblock}{Step 3}{\small Currently active}\end{alertblock}}%
    \end{column}
  \end{columns}
\end{frame}
