% Generic image placeholder box: \imgplaceholder{width}{height}{label}
\newcommand{\imgplaceholder}[3]{%
  \begin{tikzpicture}
    \draw[mcmPrimary, line width=0.8pt, fill=mcmPrimaryMuted]
      (0,0) rectangle (#1,#2);
    \draw[mcmPrimary!50, line width=0.6pt] (0,0) -- (#1,#2);
    \draw[mcmPrimary!50, line width=0.6pt] (0,#2) -- (#1,0);
    \node[fill=mcmBg, text=mcmPrimary, font=\small\bfseries, inner sep=4pt]
      at (0.5*#1,0.5*#2) {#3};
  \end{tikzpicture}%
}

% Stat badge: \statbadge{number}{label}
\newcommand{\statbadge}[2]{%
  \begin{tikzpicture}
    \node[fill=mcmPrimary, text=mcmPrimaryFg, circle, minimum size=1.7cm,
          font=\large\bfseries] (n) at (0,0) {#1};
    \node[below=0.15cm of n, text width=2.4cm, align=center,
          font=\small, text=mcmBlack] {#2};
  \end{tikzpicture}%
}

% Numbered showcase card: \showcard{number}{title}{text}
\newcommand{\showcard}[3]{%
  \begin{tikzpicture}
    \node[draw=mcmPrimary, line width=0.8pt, rounded corners=3pt,
          fill=mcmBg, text width=3.4cm, inner sep=8pt, align=left]
      (card) {%
        \tikz[baseline=-0.6ex]{%
          \node[fill=mcmPip, text=mcmPrimaryFg, circle,
                minimum size=0.55cm, font=\scriptsize\bfseries] {#1};%
        }%
        \hspace{0.3em}\textbf{#2}\\[0.3em]
        {\small #3}%
      };
  \end{tikzpicture}%
}

% #1 = index (1,2,3) of the box to highlight
\newcommand{\flowdiagram}[1]{%
  \begin{tikzpicture}[
      box/.style={draw=mcmPrimary!50, line width=0.8pt, rounded corners=3pt,
                   fill=mcmPrimary!8, minimum width=3.1cm,
                   minimum height=1.6cm, align=center, font=\small,
                   text=mcmPrimary!50},
      boxon/.style={draw=mcmPrimary, line width=2pt, rounded corners=3pt,
                     fill=mcmPrimaryMuted, minimum width=3.1cm,
                     minimum height=1.6cm, align=center, font=\small\bfseries,
                     text=mcmBlack},
      arr/.style={-{Stealth[length=2.2mm]}, mcmPrimary, line width=1pt}
    ]
    \ifnum#1=1 \tikzset{sa/.style=boxon}\else\tikzset{sa/.style=box}\fi
    \ifnum#1=2 \tikzset{sb/.style=boxon}\else\tikzset{sb/.style=box}\fi
    \ifnum#1=3 \tikzset{sc/.style=boxon}\else\tikzset{sc/.style=box}\fi
    \node[sa] (a) at (0,0) {Step 1\\{\scriptsize Input}};
    \node[sb] (b) at (4.2,0) {Step 2\\{\scriptsize Process}};
    \node[sc] (c) at (8.4,0) {Step 3\\{\scriptsize Output}};
    \draw[arr] (a.east) -- (b.west);
    \draw[arr] (b.east) -- (c.west);
  \end{tikzpicture}%
}
